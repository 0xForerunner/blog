\section{Research Experience}

\subentry{Doctoral Research: nEXO Collaboration}{2016 -- 2021}{University of British Columbia}
\begin{itemize}
  \item Designed and prototyped a Cherenkov light detector for nEXO, a neutrinoless double-beta decay search,
        built from silicon photomultipliers surrounding a cryogenic liquid-xenon core.
  \item Built a full \textbf{GEANT4 Monte Carlo} model from scratch, simulating xenon excitation and decay and
        tracking individual quanta through the detector; used it to optimize detector performance.
  \item Developed bespoke \textbf{machine-learning} algorithms to separate signal from noise in high-volume
        simulated data and quantify detector efficacy.
\end{itemize}

\subentry{Novel Photodetector Design}{2015 -- 2017}{University of Alberta}
\begin{itemize}
  \item Designed and simulated a PMT-style photodetector using a silicon photodiode array to directly detect
        photoelectrons; modelled electron-focusing fields in COMSOL and system response in GEANT4.
\end{itemize}

\subentry{PINGU Neutrino Observatory (IceCube Collaboration)}{2015 -- 2016}{University of Alberta}
\begin{itemize}
  \item Designed, built, and tested a prototype circuit board digitally controlling a picosecond LED pulser
        for precision detector calibration.
\end{itemize}

\subentry{DEAP-3600 Dark Matter Experiment}{2011 -- 2015}{University of Alberta}
\begin{itemize}
  \item Simulated, designed, built, and installed a scintillating-fibre veto photodetector to reject
        background events in the main vessel.
  \item Modelled the full detector with the Reactor Analysis Tool (RAT) to characterize background discrimination.
  \item Rewrote the data-analysis software used to process and visualize raw detector data.
  \item Tested and optimized signal-conditioning electronics on the UofA testbed; assembled detector components.
\end{itemize}
